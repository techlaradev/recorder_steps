from pytest_bdd import given, when, then, scenarios
from playwright.sync_api import Page, expect

scenarios("../login.feature")

@given("the user is on the login page")
def open_login_page(page: Page):
    page.goto("https://example.com")

@when("the user fills valid credentials")
def fill_valid_credentials(page: Page):
    page.get_by_test_id("username").fill("tester")
    page.get_by_test_id("password").fill("password")

@when("the user submits the login form")
def submit_login_form(page: Page):
    page.get_by_test_id("submit").click()

@then("the user should access the authenticated area")
def validate_authenticated_area(page: Page):
    expect(page.get_by_test_id("logout-button")).to_be_visible()
